function SubRx([string]$old, [string]$new) {
  $pat = ($old -split "`n" | ForEach-Object { [regex]::Escape($_.TrimEnd("`r")) }) -join "\r?\n"
  $c = ([regex]::Matches($script:s, $pat)).Count
  if ($c -ne 1) { throw "regex matched $c times: $($old.Substring(0,[Math]::Min(70,$old.Length)))" }
  $script:s = [regex]::Replace($script:s, $pat, { param($m) $new })
  $script:n++
}

// v10.33: THE NEXT PIECE. How far a locked piece is from being earned, as the
// counter it waits on, where it stands, and what it needs.
function cosDist(c){
  if(!c||c.how==='always'||typeof P==='undefined'||!P) return null;
  var bits=String(c.how).split(':'),k=bits[0],v=+bits[1];
  if(k==='runs')     return {k:k,cur:P.runs||0,need:v,unit:'raid'};
  if(k==='extracts') return {k:k,cur:P.ext||0,need:v,unit:'extraction'};
  if(k==='level')    return {k:k,cur:P.xpLevel||1,need:v,unit:'level'};
  if(k==='warden')   return {k:k,cur:(P.kills&&P.kills.warden)||0,need:v,unit:'Warden'};
  if(k==='board')    return {k:k,cur:(P.spClaimed||[]).length,need:SEASON_TIERS.length,unit:'board reward'};
  return null;
}
// The locked piece closest to earned: the smallest share of its gate still to
// go, ties to the fewer left. Null when the racks are complete.
function cosNext(){
  var best=null,bs=9;
  for(var i=0;i<COSMETICS.length;i++){
    var c=COSMETICS[i]; if(cosOwned(c)) continue;
    var d=cosDist(c); if(!d||!d.need) continue;
    var left=Math.max(0,d.need-d.cur), sc=left/d.need+left*1e-4;
    if(sc<bs){ bs=sc; best={c:c,d:d,left:left}; }
  }
  return best;
}
function cosNextLine(){
  var nx=cosNext();
  if(!nx) return 'THE RACKS ARE COMPLETE. EVERYTHING HERE IS YOURS.';
  var d=nx.d, left=nx.left, what;
  if(d.k==='level') what='LEVEL '+d.need+' ('+left+' TO GO)';
  else if(d.k==='board') what=left+' MORE BOARD REWARD'+(left===1?'':'S');
  else what=left+' MORE '+d.unit.toUpperCase()+(left===1?'':'S');
  var rack=(function(k){ var G={hat:'HEADGEAR',hair:'HAIR',cut:'HAIRSTYLE',beard:'BEARD',eyes:'EYES',face:'FACE',tattoo:'TATTOO',build:'BUILD',skin:'SKIN',fit:'CLOTHING',patch:'PATCH',gloves:'GLOVES',boots:'BOOTS',pack:'BACKPACK'}; return G[k]||k.toUpperCase(); })(nx.c.kind);
  return 'NEXT ON THE RACKS: '+nx.c.name.toUpperCase()+' ('+rack+'), '+what+'.';
}
